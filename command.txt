# DeepSeek V3.2 SFA manual validation commands.
#
# Run these commands manually, one by one. This file is not meant to be
# executed as a shell script.
#
# Assumptions:
#   repo:   /home/w00916487/nano-vllm-ascend-KVstar-DSA/nano-vllm-ascend
#   model:  /home/models/Deepseek-V3.2-Pruned-95B-BF/
#   cards:  0,1,2,3
#   tp:     4


# Setup. Enter repo and prepare log dir.
cd /home/w00916487/nano-vllm-ascend-KVstar-DSA/nano-vllm-ascend
mkdir -p logs/sfa_manual


# 00. Syntax check.
python -m py_compile nanovllm/models/deepseek_v32.py nanovllm/engine/model_runner.py nanovllm/engine/llm_engine.py scripts/probe_deepseek_v32_sfa_attention.py example/example.py ../ut_ops/perf_mla_prefill.py 2>&1 | tee logs/sfa_manual/00_py_compile.log


# 01. Standalone prefill: npu_lightning_indexer + npu_sparse_flash_attention.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python scripts/probe_deepseek_v32_sfa_attention.py --phase prefill --seq-len 2048 --batch-size 1 --device npu:0 2>&1 | tee logs/sfa_manual/01_probe_prefill_2k.log


# 02. Standalone decode: npu_lightning_indexer + npu_sparse_flash_attention.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python scripts/probe_deepseek_v32_sfa_attention.py --phase decode --seq-len 2048 --batch-size 4 --device npu:0 2>&1 | tee logs/sfa_manual/02_probe_decode_2k_bs4.log


# 03. Full-model short prefill correctness check.
# Keep compare enabled only for short prompts; it also runs the PyTorch MLA reference.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 NANOVLLM_MODEL=/home/models/Deepseek-V3.2-Pruned-95B-BF/ NANOVLLM_TP_SIZE=4 NANOVLLM_ENABLE_NPU_SFA_PREFILL=1 NANOVLLM_COMPARE_NPU_SFA_PREFILL=1 NANOVLLM_NPU_SFA_SPARSE_COUNT=256 NANOVLLM_LOG_NPU_SFA_INPUTS=1 NANOVLLM_LOG_NPU_SFA_TIMING=1 NANOVLLM_PROFILE_LAYER_IDS=0 NANOVLLM_MAX_MODEL_LEN=256 NANOVLLM_MAX_BATCHED_TOKENS=512 NANOVLLM_MAX_NUM_SEQS=2 NANOVLLM_MAX_GEN_TOKENS=1 NANOVLLM_SKIP_WARMUP=1 python example/example.py 2>&1 | tee logs/sfa_manual/03_short_prefill_compare.log


# 04. Full-model long path: one exact 2048-token prompt, SFA prefill + SFA decode, generate 1 token.
# Do not enable compare here, otherwise the dense MLA reference will dominate runtime.
# Keep NANOVLLM_MAX_BATCHED_TOKENS >= NANOVLLM_MAX_MODEL_LEN, otherwise nanovllm rejects the config.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 NANOVLLM_MODEL=/home/models/Deepseek-V3.2-Pruned-95B-BF/ NANOVLLM_TP_SIZE=4 NANOVLLM_ENABLE_NPU_SFA_PREFILL=1 NANOVLLM_ENABLE_NPU_SFA_DECODE=1 NANOVLLM_NPU_SFA_SPARSE_COUNT=2048 NANOVLLM_LOG_NPU_SFA_INPUTS=1 NANOVLLM_LOG_NPU_SFA_TIMING=1 NANOVLLM_DUMP_NPU_SFA_INPUTS=logs/sfa_manual/dumps NANOVLLM_DUMP_NPU_SFA_MAX_CALLS=1 NANOVLLM_PROFILE_LAYER_IDS=0 NANOVLLM_LONG_PROMPT_TOKENS=2048 NANOVLLM_MAX_MODEL_LEN=2176 NANOVLLM_MAX_BATCHED_TOKENS=3072 NANOVLLM_MAX_NUM_SEQS=1 NANOVLLM_MAX_GEN_TOKENS=1 NANOVLLM_SKIP_WARMUP=1 python example/example.py 2>&1 | tee logs/sfa_manual/04_long_prefill_sfa_decode1.log


# 05. If step 04 created dumps, pick one actual path and replay it on one NPU.
# This checks whether one dumped SFA input, including causal -1 sparse_indices,
# can run npu_sparse_flash_attention by itself.
find logs/sfa_manual/dumps -name 'sfa_rank*_layer*_seq*.pt' | head -n 5
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python scripts/probe_deepseek_v32_sfa_attention.py --replay-dump logs/sfa_manual/dumps/REPLACE_WITH_ACTUAL_DUMP.pt --device npu:0 2>&1 | tee logs/sfa_manual/05_replay_dump.log


# 06. Replay one dumped SFA input per rank concurrently.
# This checks whether the same dumped SFA inputs only fail when TP ranks call
# npu_sparse_flash_attention at the same time.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python scripts/probe_npu_sfa_decode.py --dump-dir logs/sfa_manual/dumps --world-size 4 --hccl-port 28089 --repeats 1 2>&1 | tee logs/sfa_manual/06_replay_dump_dir_tp4.log


# 07. Trace the working vllm-ascend 0.19 SFA path and print real op inputs.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 SFA_TRACE_MAX_CALLS=8 python scripts/trace_vllm_ascend_sfa_inputs.py --model /home/models/Deepseek-V3.2-Pruned-95B-BF/ --tp-size 4 --prompt-tokens 2048 --max-model-len 2176 --max-num-batched-tokens 3072 --max-tokens 1 2>&1 | tee logs/sfa_manual/07_vllm_ascend_sfa_trace.log


# 08. Full Nano SFA path without dumping tensors.
# This checks whether NANOVLLM_DUMP_NPU_SFA_INPUTS itself changes the crash behavior.
PYTORCH_NPU_ALLOC_CONF=expandable_segments:True ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 NANOVLLM_MODEL=/home/models/Deepseek-V3.2-Pruned-95B-BF/ NANOVLLM_TP_SIZE=4 NANOVLLM_ENABLE_NPU_SFA_PREFILL=1 NANOVLLM_ENABLE_NPU_SFA_DECODE=1 NANOVLLM_NPU_SFA_SPARSE_COUNT=2048 NANOVLLM_LOG_NPU_SFA_INPUTS=1 NANOVLLM_LOG_NPU_SFA_TIMING=1 NANOVLLM_NPU_SFA_STRICT=1 NANOVLLM_PROFILE_LAYER_IDS=0 NANOVLLM_LONG_PROMPT_TOKENS=2048 NANOVLLM_MAX_MODEL_LEN=2176 NANOVLLM_MAX_BATCHED_TOKENS=3072 NANOVLLM_MAX_NUM_SEQS=1 NANOVLLM_MAX_GEN_TOKENS=1 NANOVLLM_SKIP_WARMUP=1 python example/example.py 2>&1 | tee logs/sfa_manual/08_nano_full_sfa_no_dump_2048.log


# 09. Re-run the vllm-ascend 0.19 trace after stride/storage_offset logging is enabled.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 SFA_TRACE_MAX_CALLS=2 python scripts/trace_vllm_ascend_sfa_inputs.py --model /home/models/Deepseek-V3.2-Pruned-95B-BF/ --tp-size 4 --prompt-tokens 2048 --max-model-len 2176 --max-num-batched-tokens 3072 --max-tokens 1 2>&1 | tee logs/sfa_manual/09_vllm_ascend_sfa_trace_stride.log


# 10. Re-run the full Nano SFA path after aligning with vllm-ascend's
# torch.npu.config.allow_internal_format=True setting.
PYTORCH_NPU_ALLOC_CONF=expandable_segments:True ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 NANOVLLM_MODEL=/home/models/Deepseek-V3.2-Pruned-95B-BF/ NANOVLLM_TP_SIZE=4 NANOVLLM_ENABLE_NPU_SFA_PREFILL=1 NANOVLLM_ENABLE_NPU_SFA_DECODE=1 NANOVLLM_NPU_SFA_SPARSE_COUNT=2048 NANOVLLM_LOG_NPU_SFA_INPUTS=1 NANOVLLM_LOG_NPU_SFA_TIMING=1 NANOVLLM_NPU_SFA_STRICT=1 NANOVLLM_PROFILE_LAYER_IDS=0 NANOVLLM_LONG_PROMPT_TOKENS=2048 NANOVLLM_MAX_MODEL_LEN=2176 NANOVLLM_MAX_BATCHED_TOKENS=3072 NANOVLLM_MAX_NUM_SEQS=1 NANOVLLM_MAX_GEN_TOKENS=1 NANOVLLM_SKIP_WARMUP=1 python example/example.py 2>&1 | tee logs/sfa_manual/10_nano_full_sfa_internal_format_2048.log


# 11. Short prompt sanity check with the same internal-format setting.
PYTORCH_NPU_ALLOC_CONF=expandable_segments:True ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 NANOVLLM_MODEL=/home/models/Deepseek-V3.2-Pruned-95B-BF/ NANOVLLM_TP_SIZE=4 NANOVLLM_ENABLE_NPU_SFA_PREFILL=1 NANOVLLM_ENABLE_NPU_SFA_DECODE=1 NANOVLLM_NPU_SFA_SPARSE_COUNT=2048 NANOVLLM_LOG_NPU_SFA_INPUTS=1 NANOVLLM_LOG_NPU_SFA_TIMING=1 NANOVLLM_NPU_SFA_STRICT=1 NANOVLLM_PROFILE_LAYER_IDS=0 NANOVLLM_LONG_PROMPT_TOKENS=129 NANOVLLM_MAX_MODEL_LEN=256 NANOVLLM_MAX_BATCHED_TOKENS=256 NANOVLLM_MAX_NUM_SEQS=1 NANOVLLM_MAX_GEN_TOKENS=1 NANOVLLM_SKIP_WARMUP=1 python example/example.py 2>&1 | tee logs/sfa_manual/11_nano_full_sfa_internal_format_129.log


# 12. Dense MLA prefill op probe, small enough to print a CPU reference diff.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python ../ut_ops/perf_mla_prefill.py --device npu:0 --seq-lens 128 --mask-size 2048 --warmup 1 --iters 5 --check-reference 2>&1 | tee logs/sfa_manual/12_dense_mla_prefill_128.log


# 13. Dense MLA prefill op probe, DeepSeek V3.2 TP=4 local-head shape and 2048 tokens.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python ../ut_ops/perf_mla_prefill.py --device npu:0 --seq-lens 2048 --mask-size 2048 --warmup 3 --iters 10 2>&1 | tee logs/sfa_manual/13_dense_mla_prefill_2048.log


# 14. Dense MLA prefill op probe above 2k tokens. If this fails because of mask size
# or FIA shape limits, keep 13 as the first integration target and we will add chunking.
ASCEND_RT_VISIBLE_DEVICES=0,1,2,3 python ../ut_ops/perf_mla_prefill.py --device npu:0 --seq-lens 2176 --mask-size 4096 --warmup 1 --iters 3 2>&1 | tee logs/sfa_manual/14_dense_mla_prefill_2176.log


# 99. Extract key lines.
grep -E "PROBE |MLA_PROBE|MLA_BENCH|SFA_TRACE|built long prompt|TP-wide minimum KV cache blocks|NPU SFA input summary|NPU SFA timing|NPU SFA compare|NPU SFA .*failed|Dumped NPU SFA|Traceback|RuntimeError|ERROR|Prompt:|prompt len:|text:" logs/sfa_manual/*.log 2>&1 | tee logs/sfa_manual/99_key_lines.log


# Paste these logs back.
cat logs/sfa_manual/01_probe_prefill_2k.log
cat logs/sfa_manual/02_probe_decode_2k_bs4.log
cat logs/sfa_manual/03_short_prefill_compare.log
cat logs/sfa_manual/04_long_prefill_sfa_decode1.log
cat logs/sfa_manual/05_replay_dump.log
cat logs/sfa_manual/06_replay_dump_dir_tp4.log
cat logs/sfa_manual/07_vllm_ascend_sfa_trace.log
cat logs/sfa_manual/08_nano_full_sfa_no_dump_2048.log
cat logs/sfa_manual/09_vllm_ascend_sfa_trace_stride.log
cat logs/sfa_manual/10_nano_full_sfa_internal_format_2048.log
cat logs/sfa_manual/11_nano_full_sfa_internal_format_129.log
cat logs/sfa_manual/12_dense_mla_prefill_128.log
cat logs/sfa_manual/13_dense_mla_prefill_2048.log
cat logs/sfa_manual/14_dense_mla_prefill_2176.log
cat logs/sfa_manual/99_key_lines.log
