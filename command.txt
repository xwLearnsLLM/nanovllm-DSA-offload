# Run these commands one by one on the Ascend machine from the directory that
# contains run.py. Do not run all benchmark variants in one batch; stop and send
# the log if any command prints "NPU SFA decode fast path failed once".

mkdir -p logs

# 0) Probe whether the vllm-ascend-style custom ops exist in this environment.
python - <<'PY' 2>&1 | tee logs/00_op_probe.log
import torch

print("torch:", getattr(torch, "__version__", "unknown"))
try:
    import torch_npu
    print("torch_npu import: ok")
except Exception as exc:
    torch_npu = None
    print("torch_npu import: failed", repr(exc))

if hasattr(torch, "npu"):
    try:
        print("npu_count:", torch.npu.device_count())
    except Exception as exc:
        print("npu_count: failed", repr(exc))

ascend_ops = getattr(torch.ops, "_C_ascend", None)
for name in (
    "moe_gating_top_k",
    "npu_lightning_indexer",
    "npu_sparse_flash_attention",
    "batch_matmul_transpose",
    "mla_preprocess",
):
    print(f"torch.ops._C_ascend.{name}:", bool(ascend_ops is not None and hasattr(ascend_ops, name)))

if torch_npu is not None:
    for name in (
        "npu_moe_gating_top_k",
        "npu_lightning_indexer",
        "npu_kv_rmsnorm_rope_cache",
        "npu_interleave_rope",
    ):
        print(f"torch_npu.{name}:", hasattr(torch_npu, name))
PY

# 1) Baseline after the current nano patch. This keeps SFA and NPU MoE gating off.
NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=1 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_DECODE=1 python run.py 2>&1 | tee logs/01_default_decode.log

# 2) Baseline layer split. Paste the "Layer profile avg" lines back.
NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=0 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_LAYERS=1 NANOVLLM_PROFILE_LAYER_IDS=0,mid,last python run.py 2>&1 | tee logs/02_default_layers.log

# 3) Enable the SFA decode fast path: npu_lightning_indexer + npu_sparse_flash_attention.
NANOVLLM_ENABLE_NPU_SFA=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=1 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_DECODE=1 python run.py 2>&1 | tee logs/03_sfa_decode.log

# 4) SFA layer split. The attention column should drop if the fast path is active.
NANOVLLM_ENABLE_NPU_SFA=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=0 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_LAYERS=1 NANOVLLM_PROFILE_LAYER_IDS=0,mid,last python run.py 2>&1 | tee logs/04_sfa_layers.log

# 5) Enable SFA plus NPU MoE gating. This only optimizes router top-k, not expert MLP yet.
NANOVLLM_ENABLE_NPU_SFA=1 NANOVLLM_ENABLE_NPU_MOE_GATING=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=1 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_DECODE=1 python run.py 2>&1 | tee logs/05_sfa_moe_gate_decode.log

# 6) SFA plus NPU MoE gating layer split.
NANOVLLM_ENABLE_NPU_SFA=1 NANOVLLM_ENABLE_NPU_MOE_GATING=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=0 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_LAYERS=1 NANOVLLM_PROFILE_LAYER_IDS=0,mid,last python run.py 2>&1 | tee logs/06_sfa_moe_gate_layers.log

# 7) Optional isolation run: only use this if the default patched baseline is
# slower than your previous 9-12 tok/s result. It disables batch_matmul_transpose.
NANOVLLM_DISABLE_BATCH_MATMUL_TRANSPOSE=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=1 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_DECODE=1 python run.py 2>&1 | tee logs/07_no_batch_matmul_transpose_decode.log

# 8) After the runs, this extracts the lines I need most.
grep -E "Decode profile avg|Layer profile avg|Perf summary|summary :|NPU SFA decode fast path failed|response:" logs/*.log
