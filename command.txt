# 0) Probe the custom ops. This was skipped last time, and it tells us whether
# the SFA path can really call the vllm-ascend-style kernels.
mkdir -p logs && python - <<'PY' 2>&1 | tee logs/10_op_probe.log
import torch

print("torch:", getattr(torch, "__version__", "unknown"))
try:
    import torch_npu
    print("torch_npu import: ok")
except Exception as exc:
    torch_npu = None
    print("torch_npu import: failed", repr(exc))

if hasattr(torch, "npu"):
    try:
        print("npu_count:", torch.npu.device_count())
    except Exception as exc:
        print("npu_count: failed", repr(exc))

ascend_ops = getattr(torch.ops, "_C_ascend", None)
for name in (
    "moe_gating_top_k",
    "npu_lightning_indexer",
    "npu_sparse_flash_attention",
    "batch_matmul_transpose",
    "mla_preprocess",
):
    print(f"torch.ops._C_ascend.{name}:", bool(ascend_ops is not None and hasattr(ascend_ops, name)))

if torch_npu is not None:
    for name in (
        "npu_moe_gating_top_k",
        "npu_lightning_indexer",
        "npu_kv_rmsnorm_rope_cache",
        "npu_interleave_rope",
    ):
        print(f"torch_npu.{name}:", hasattr(torch_npu, name))
PY

# 1) Confirm whether NANOVLLM_ENABLE_NPU_SFA=1 is actually active or silently
# falling back. Look for "NPU SFA decode fast path active/unavailable/failed".
mkdir -p logs && NANOVLLM_ENABLE_NPU_SFA=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=1 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_DECODE=1 python run.py 2>&1 | tee logs/11_sfa_status_decode.log

# 2) Detailed split for SFA enabled. Paste the "Attention detail avg",
# "MoE detail avg", and "Layer profile avg" lines.
mkdir -p logs && NANOVLLM_ENABLE_NPU_SFA=1 NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=0 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_LAYERS=1 NANOVLLM_PROFILE_ATTENTION_DETAIL=1 NANOVLLM_PROFILE_MOE_DETAIL=1 NANOVLLM_PROFILE_LAYER_IDS=mid,last python run.py 2>&1 | tee logs/12_sfa_detail.log

# 3) Detailed split for the native DSA path. This compares against step 2 and
# shows exactly what the SFA switch changed.
mkdir -p logs && NANOVLLM_USE_TQDM=0 NANOVLLM_WARMUP_RUNS=0 NANOVLLM_BENCH_RUNS=1 NANOVLLM_PERF_SUMMARY=1 NANOVLLM_PROFILE_LAYERS=1 NANOVLLM_PROFILE_ATTENTION_DETAIL=1 NANOVLLM_PROFILE_MOE_DETAIL=1 NANOVLLM_PROFILE_LAYER_IDS=mid,last python run.py 2>&1 | tee logs/13_native_detail.log

# 4) Extract the important lines after the three runs.
grep -E "NPU SFA decode fast path|Decode profile avg|Attention detail avg|MoE detail avg|Layer profile avg|Perf summary|summary :" logs/10_op_probe.log logs/11_sfa_status_decode.log logs/12_sfa_detail.log logs/13_native_detail.log
