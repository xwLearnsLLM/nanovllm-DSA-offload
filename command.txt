# 1) Install a temporary sitecustomize hook. PYTHONPATH injects it into vllm
# worker processes too, so we can see the real op arguments used by
# vllm-ascend 0.15 on that machine, independent of the local 0.14 source tree.
rm -rf /tmp/nanovllm_vllm_probe
mkdir -p /tmp/nanovllm_vllm_probe
cat >/tmp/nanovllm_vllm_probe/sitecustomize.py <<'PY'
import os

if os.environ.get("VLLM_ASCEND_SFA_PROBE") == "1":
    try:
        import torch
        import torch.distributed as dist
        import torch_npu
        import vllm
        import vllm_ascend
        from vllm_ascend import vllm_ascend_C  # noqa: F401

        max_prints = int(os.environ.get("VLLM_ASCEND_SFA_PROBE_MAX_PRINTS", "4"))
        only_decode = os.environ.get("VLLM_ASCEND_SFA_PROBE_ONLY_DECODE", "0").lower() in (
            "1",
            "true",
            "yes",
            "on",
        )
        counts = {}

        def rank_info():
            try:
                if dist.is_available() and dist.is_initialized():
                    return dist.get_rank()
            except Exception:
                pass
            return "na"

        def tinfo(x):
            if x is None:
                return "None"
            return f"shape={tuple(x.shape)} dtype={x.dtype} device={x.device}"

        def head(x, n=8):
            if x is None or not hasattr(x, "numel") or x.numel() == 0:
                return []
            try:
                y = x
                if y.dim() >= 2:
                    y = y[0]
                return y.flatten()[:n].detach().cpu().tolist()
            except Exception as exc:
                return f"head_failed={exc!r}"

        def should_print(key):
            count = counts.get(key, 0)
            if count >= max_prints:
                return False
            counts[key] = count + 1
            return True

        def is_decode_like(*args, **kwargs):
            actual_seq_lengths_query = kwargs.get("actual_seq_lengths_query")
            query = kwargs.get("query", args[0] if len(args) > 0 else None)
            if actual_seq_lengths_query is None:
                return False
            try:
                num_reqs = actual_seq_lengths_query.numel()
                return (
                    query is not None
                    and query.shape[0] == num_reqs
                    and int(actual_seq_lengths_query[-1].item()) == num_reqs
                )
            except Exception:
                return False

        def log_indexer(source, *args, **kwargs):
            query = kwargs.get("query", args[0] if len(args) > 0 else None)
            key = kwargs.get("key", args[1] if len(args) > 1 else None)
            weights = kwargs.get("weights", args[2] if len(args) > 2 else None)
            aqlq = kwargs.get("actual_seq_lengths_query")
            aqlk = kwargs.get("actual_seq_lengths_key")
            block_table = kwargs.get("block_table")
            print(
                "VLLM_SFA_PROBE indexer "
                f"source={source} pid={os.getpid()} rank={rank_info()} "
                f"query={tinfo(query)} key={tinfo(key)} weights={tinfo(weights)} "
                f"aqlq={tinfo(aqlq)} aqlq_head={head(aqlq)} "
                f"aqlk={tinfo(aqlk)} aqlk_head={head(aqlk)} "
                f"block_table={tinfo(block_table)} block_table0_head={head(block_table)} "
                f"sparse_count={kwargs.get('sparse_count')} sparse_mode={kwargs.get('sparse_mode')} "
                f"layout_query={kwargs.get('layout_query')} layout_key={kwargs.get('layout_key')}",
                flush=True,
            )

        def log_sfa(source, *args, **kwargs):
            query = kwargs.get("query", args[0] if len(args) > 0 else None)
            key = kwargs.get("key", args[1] if len(args) > 1 else None)
            value = kwargs.get("value", args[2] if len(args) > 2 else None)
            sparse_indices = kwargs.get("sparse_indices", args[3] if len(args) > 3 else None)
            aqlq = kwargs.get("actual_seq_lengths_query")
            aqlkv = kwargs.get("actual_seq_lengths_kv")
            block_table = kwargs.get("block_table")
            query_rope = kwargs.get("query_rope")
            key_rope = kwargs.get("key_rope")
            print(
                "VLLM_SFA_PROBE sfa "
                f"source={source} pid={os.getpid()} rank={rank_info()} "
                f"query={tinfo(query)} key={tinfo(key)} value={tinfo(value)} "
                f"sparse_indices={tinfo(sparse_indices)} sparse_head={head(sparse_indices)} "
                f"aqlq={tinfo(aqlq)} aqlq_head={head(aqlq)} "
                f"aqlkv={tinfo(aqlkv)} aqlkv_head={head(aqlkv)} "
                f"block_table={tinfo(block_table)} block_table0_head={head(block_table)} "
                f"query_rope={tinfo(query_rope)} key_rope={tinfo(key_rope)} "
                f"sparse_block_size={kwargs.get('sparse_block_size')} sparse_mode={kwargs.get('sparse_mode')} "
                f"layout_query={kwargs.get('layout_query')} layout_kv={kwargs.get('layout_kv')}",
                flush=True,
            )

        def wrap_indexer(owner, attr, source):
            if not hasattr(owner, attr):
                print(f"VLLM_SFA_PROBE missing source={source} attr={attr}", flush=True)
                return
            original = getattr(owner, attr)

            def wrapped(*args, **kwargs):
                if only_decode and not is_decode_like(*args, **kwargs):
                    return original(*args, **kwargs)
                if should_print((source, "indexer")):
                    log_indexer(source, *args, **kwargs)
                return original(*args, **kwargs)

            setattr(owner, attr, wrapped)
            print(f"VLLM_SFA_PROBE wrapped source={source} attr={attr}", flush=True)

        def wrap_sfa(owner, attr, source):
            if not hasattr(owner, attr):
                print(f"VLLM_SFA_PROBE missing source={source} attr={attr}", flush=True)
                return
            original = getattr(owner, attr)

            def wrapped(*args, **kwargs):
                if only_decode and not is_decode_like(*args, **kwargs):
                    return original(*args, **kwargs)
                if should_print((source, "sfa")):
                    log_sfa(source, *args, **kwargs)
                return original(*args, **kwargs)

            setattr(owner, attr, wrapped)
            print(f"VLLM_SFA_PROBE wrapped source={source} attr={attr}", flush=True)

        print(
            "VLLM_SFA_PROBE installed "
            f"pid={os.getpid()} rank={rank_info()} "
            f"torch={getattr(torch, '__version__', 'unknown')} "
            f"vllm={getattr(vllm, '__version__', 'unknown')} "
            f"only_decode={only_decode} "
            f"vllm_file={getattr(vllm, '__file__', 'unknown')} "
            f"vllm_ascend_file={getattr(vllm_ascend, '__file__', 'unknown')}",
            flush=True,
        )

        ascend_ops = getattr(torch.ops, "_C_ascend", None)
        if ascend_ops is not None:
            wrap_indexer(ascend_ops, "npu_lightning_indexer", "torch.ops._C_ascend")
            wrap_sfa(ascend_ops, "npu_sparse_flash_attention", "torch.ops._C_ascend")
        else:
            print("VLLM_SFA_PROBE missing source=torch.ops._C_ascend", flush=True)

        wrap_indexer(torch_npu, "npu_lightning_indexer", "torch_npu")
        wrap_sfa(torch_npu, "npu_sparse_flash_attention", "torch_npu")
    except Exception as exc:
        print(f"VLLM_SFA_PROBE install_failed pid={os.getpid()} error={exc!r}", flush=True)
PY

import os
from time import perf_counter

from nanovllm import LLM, SamplingParams


MODEL_PATH = "/home/models/Deepseek-V3.2-Pruned-95B-BF/"
TP_SIZE = 4


def main():
    max_model_len = int(os.environ.get("PROBE_MAX_MODEL_LEN", "4096"))
    prompts = [
        "The capital city of China is",
        "calculate 2 + 3 * 4 = ",
        "List all prime numbers < 100: 2, 3, ",
        "1, 3, 7, 15, 31, ",
        "Large Language Model is a type of AI model that",
        "If Tom is shorter than Jerry, Spike is taller than Tom, then the shortest person is",
        "If a train leaves at 2 PM and takes 3 hours to arrive, it will arrive at",
    ]

    llm = LLM(
        model=MODEL_PATH,
        enforce_eager=True,
        tensor_parallel_size=TP_SIZE,
        enable_expert_parallel=True,
        max_model_len=max_model_len,
        max_num_batched_tokens=max_model_len,
        max_num_seqs=128,
        kvcache_block_size=128,
        skip_warmup=1,
        trust_remote_code=True,
        gpu_memory_utilization=0.95,
    )

    start = perf_counter()
    outputs = llm.generate(
        prompts,
        SamplingParams(temperature=0.02, max_tokens=32),
        use_tqdm=False,
    )
    elapsed = max(perf_counter() - start, 1e-9)
    total_output_tokens = sum(len(output["token_ids"]) for output in outputs)
    print(
        "summary :",
        f"run 1/1, elapsed={elapsed:.2f}s, output={total_output_tokens / elapsed:.2f} toks/s",
    )
    for prompt, output in zip(prompts, outputs):
        print("prompt  :", prompt)
        print("response:", repr(output["text"]))


if __name__ == "__main__":
    main()
PY
