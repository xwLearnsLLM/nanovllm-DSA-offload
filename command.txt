# 1) Show whether importing vllm_ascend_C registers the two SFA custom ops.
mkdir -p logs && python - <<'PY' 2>&1 | tee logs/30_vllm_ascend_op_import_probe.log
import torch

def show(label):
    ascend_ops = getattr(torch.ops, "_C_ascend", None)
    print(f"== {label} ==")
    for name in (
        "npu_lightning_indexer",
        "npu_sparse_flash_attention",
        "batch_matmul_transpose",
        "moe_gating_top_k",
    ):
        print(f"torch.ops._C_ascend.{name}:", bool(ascend_ops is not None and hasattr(ascend_ops, name)))
    try:
        import torch_npu
        for name in (
            "npu_lightning_indexer",
            "npu_sparse_flash_attention",
            "npu_kv_rmsnorm_rope_cache",
            "npu_interleave_rope",
            "npu_moe_gating_top_k",
        ):
            print(f"torch_npu.{name}:", hasattr(torch_npu, name))
    except Exception as exc:
        print("torch_npu import failed:", repr(exc))

show("before vllm_ascend_C import")
try:
    import vllm
    import vllm_ascend
    from vllm_ascend import vllm_ascend_C
    print("vllm_ascend_C import: ok")
except Exception as exc:
    print("vllm_ascend_C import: failed", repr(exc))
show("after vllm_ascend_C import")
PY
