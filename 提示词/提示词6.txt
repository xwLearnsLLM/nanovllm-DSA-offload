llm = LLM(

    tokenizer_mode="deepseek_v32",
    quantization="ascend",
    enable_expert_parallel=True,
    dtype="auto",   # 或 "bfloat16"，不要强制 float16

    "用Python写一个快速排序算法",
]

    top_p=0.9,

print(f"\n✓ 推理完成！")
